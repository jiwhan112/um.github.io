cmake_minimum_required(VERSION 3.16)
project(NET04_Reference LANGUAGES CXX)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 20)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(NET04 main.cpp)
if(MSVC)
  target_compile_options(NET04 PRIVATE /W4)
else()
  target_compile_options(NET04 PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
enable_testing()
add_test(NAME NET04 COMMAND NET04)

# 현재 교재가 추가한 경계 검사: 원본 main.cpp의 구현을 그대로 포함합니다.
add_executable(network_edges edge_cases.cpp)
set_target_properties(network_edges PROPERTIES CXX_STANDARD ${CMAKE_CXX_STANDARD} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(network_edges PRIVATE /W4 /EHsc)
else()
  target_compile_options(network_edges PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME network_edges COMMAND network_edges)
set_tests_properties(network_edges NET04 PROPERTIES TIMEOUT 10)
