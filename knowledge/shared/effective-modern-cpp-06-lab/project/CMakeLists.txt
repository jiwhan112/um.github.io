cmake_minimum_required(VERSION 3.16)
project(effective_modern_06 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(lesson main.cpp)
target_link_libraries(lesson PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(lesson PRIVATE /W4 /utf-8)
else()
  target_compile_options(lesson PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME lesson_contract COMMAND lesson)
