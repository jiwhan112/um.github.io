cmake_minimum_required(VERSION 3.16)
project(dx11_gs_cpu LANGUAGES CXX)
set(LAB_CXX_STANDARD 20 CACHE STRING "20 default, 11 compatible")
add_executable(dx11_14_cpu main.cpp)
set_target_properties(dx11_14_cpu PROPERTIES CXX_STANDARD ${LAB_CXX_STANDARD} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(dx11_14_cpu PRIVATE /W4 /EHsc)
else()
  target_compile_options(dx11_14_cpu PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME geometry_contract COMMAND dx11_14_cpu)
