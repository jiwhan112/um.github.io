cmake_minimum_required(VERSION 3.16)
project(os06_false_sharing LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "Use 11 or 20")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(os06_cpu main.cpp)
target_link_libraries(os06_cpu PRIVATE Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(os06_cpu PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME false_sharing_contract COMMAND os06_cpu --test)
