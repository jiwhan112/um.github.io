cmake_minimum_required(VERSION 3.16)
project(algorithm_core_06 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(lesson main.cpp)
if(MSVC)
  target_compile_options(lesson PRIVATE /W4 /utf-8)
else()
  target_compile_options(lesson PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME contract_tests COMMAND lesson)
