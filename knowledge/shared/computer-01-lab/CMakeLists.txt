cmake_minimum_required(VERSION 3.16)
project(computer_instruction_trace LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11 CACHE STRING "Teaching example standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(instruction_trace main.cpp)
if(MSVC)
  target_compile_options(instruction_trace PRIVATE /W4)
else()
  target_compile_options(instruction_trace PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME instruction_trace_contract COMMAND instruction_trace)
set_tests_properties(instruction_trace_contract PROPERTIES TIMEOUT 10)
