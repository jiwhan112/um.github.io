cmake_minimum_required(VERSION 3.16)
project(dx9_picking_recovered LANGUAGES CXX)
set(LAB_CXX_STANDARD 20 CACHE STRING "20 or 11")
add_executable(picking main.cpp)
set_target_properties(picking PROPERTIES CXX_STANDARD ${LAB_CXX_STANDARD} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(picking PRIVATE /W4 /EHsc)
else()
  target_compile_options(picking PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME picking COMMAND picking)
