# 2026-10-24 독립 다운로드 빌드. 원문 CMake는 original/에 보존합니다.
cmake_minimum_required(VERSION 3.16)
project(os_03_learning LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard: 11 or 20")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
enable_testing()
add_executable(os_03_korean main.cpp)
add_executable(os_03_captured original/main.cpp)
foreach(target os_03_korean os_03_captured)
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
  add_test(NAME ${target} COMMAND ${target})
  set_tests_properties(${target} PROPERTIES TIMEOUT 30)
endforeach()
