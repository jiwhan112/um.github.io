cmake_minimum_required(VERSION 3.16)
project(notion_cs_run6 LANGUAGES CXX)
set(LAB_CXX_STANDARD "20" CACHE STRING "11 or 20")
set(CMAKE_CXX_STANDARD ${LAB_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
enable_testing()
add_executable(os03 OS03/main.cpp)
target_link_libraries(os03 PRIVATE Threads::Threads)
add_executable(net03 NET03/main.cpp)
foreach(target IN ITEMS os03 net03)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  add_test(NAME ${target} COMMAND ${target})
endforeach()
