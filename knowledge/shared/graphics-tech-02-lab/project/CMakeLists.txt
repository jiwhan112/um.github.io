cmake_minimum_required(VERSION 3.16)
project(graphics_tech_02 LANGUAGES CXX)
add_executable(graphics_lab main.cpp)
target_compile_features(graphics_lab PRIVATE cxx_std_17)
set_target_properties(graphics_lab PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(graphics_lab PRIVATE /W4 /utf-8)
else()
  target_compile_options(graphics_lab PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME contracts COMMAND graphics_lab)
