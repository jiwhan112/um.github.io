cmake_minimum_required(VERSION 3.16)
project(net02_serialization LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(net02_serialization main.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(net02_serialization PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
enable_testing()
add_test(NAME net02_serialization COMMAND net02_serialization)
