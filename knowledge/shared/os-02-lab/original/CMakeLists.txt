cmake_minimum_required(VERSION 3.16)
project(os02_pagetable LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(os02_pagetable main.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(os02_pagetable PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
enable_testing()
add_test(NAME os02_pagetable COMMAND os02_pagetable)
