cmake_minimum_required(VERSION 3.16)
project(os_queue_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(os_queue main.cpp)
target_link_libraries(os_queue PRIVATE Threads::Threads)
enable_testing()
add_test(NAME os_queue COMMAND os_queue)
