cmake_minimum_required(VERSION 3.16)
project(modern_cpp_02 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ language standard (20 or 23)")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(modern_cpp_02 main.cpp)
target_link_libraries(modern_cpp_02 PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(modern_cpp_02 PRIVATE /W4 /utf-8 /Zc:__cplusplus)
else()
  target_compile_options(modern_cpp_02 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME modern_cpp_02_contract COMMAND modern_cpp_02)
set_tests_properties(modern_cpp_02_contract PROPERTIES TIMEOUT 10)
