cmake_minimum_required(VERSION 3.20)
project(AtlasDx12Windows LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows SDK required. CPU chapter tests are a separate project/ directory.")
endif()
add_executable(atlas_dx12_renderer WIN32 renderer.cpp)
add_executable(atlas_dx12_compute compute.cpp)
foreach(target atlas_dx12_renderer atlas_dx12_compute)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(${target} PRIVATE d3d12 dxgi d3dcompiler dxguid)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  endif()
endforeach()
