cmake_minimum_required(VERSION 3.16)
project(network_01_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(network_01_cpu main.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(network_01_cpu PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
enable_testing()
add_test(NAME network_01_cpu COMMAND network_01_cpu)
set_tests_properties(network_01_cpu PROPERTIES TIMEOUT 15)
