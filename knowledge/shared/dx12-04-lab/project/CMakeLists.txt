cmake_minimum_required(VERSION 3.16)
project(AtlasDx12Contract04 LANGUAGES CXX)
add_executable(dx12_contract main.cpp)
target_compile_features(dx12_contract PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(dx12_contract PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(dx12_contract PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME dx12_04_cpu_contract COMMAND dx12_contract)
