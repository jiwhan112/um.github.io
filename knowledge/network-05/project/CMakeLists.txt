cmake_minimum_required(VERSION 3.16)
project(NET05_mtu_budget LANGUAGES CXX)
set(LAB_CXX_STANDARD 20 CACHE STRING "C++ standard: 20 or 11")
enable_testing()
add_executable(net05 main.cpp)
set_target_properties(net05 PROPERTIES CXX_STANDARD ${LAB_CXX_STANDARD}
  CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(net05 PRIVATE /W4 /EHsc)
else()
  target_compile_options(net05 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME NET05_mtu_budget COMMAND net05)

# 현재 교재가 추가한 경계 검사: 원본 main.cpp의 구현을 그대로 포함합니다.
add_executable(network_edges edge_cases.cpp)
set_target_properties(network_edges PROPERTIES CXX_STANDARD ${LAB_CXX_STANDARD} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(network_edges PRIVATE /W4 /EHsc)
else()
  target_compile_options(network_edges PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME network_edges COMMAND network_edges)
set_tests_properties(network_edges NET05_mtu_budget PROPERTIES TIMEOUT 10)
